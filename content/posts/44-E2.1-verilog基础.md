### Lemmings 4

![image.png](https://bucket-plusone.oss-cn-wuhan-lr.aliyuncs.com/img/20261101080344419.png)
这里的module declaration：
```verilog
module top_module(
    input clk,
    input areset,    // Freshly brainwashed Lemmings walk left.
    input bump_left,
    input bump_right,
    input ground,
    input dig,
    output walk_left,
    output walk_right,
    output aaah,
    output digging ); 

endmodule

```

这里面的行为一共有三种，分别是掉落，挖掘，撞墙换向。状态有四种，掉落中，挖掘中，死亡，正常行走，只有在正常行走的时候，撞墙信号才会生效。

正常行走的逻辑：`ground== 1 && dig==0`
bump_left && bump_right：换向
bump_left:向右走
bump_right:向左走

掉落逻辑：ground=0, aaah=1
ground=1 落地，要判断瞬间的时长。所有的状态全部置为0

挖掘逻辑：ground=1&&dig=1
知道ground=0时刻，记录挖掘状态

先判断是否掉落，再判断是否挖掘，最后判断是否行走.我们先明确这里的状态转移逻辑，其中可以从挖掘/走路到掉落，也可以从掉落衔接到挖掘和走路。死亡的状态可以在这里进行判断，因为这里是一个组合逻辑电路，可以根据所记录的掉落时间来产生判断。

这里我们要用到parameter，来定义我们的一些表示状态的参量。因为这里一共2行走+2掉落走+2挖掘走+1死亡一共7个变量，可以用三个字节来表示.对于这种明显的转移的风格，可以使用case 语句

我们采用next_state和state来表达现在的状态和下一个时刻的状态，其中我们的next_state使用的是always @(`*`)

豆包帮我排错了好久拍不出，gpt一分钟就看出来了，如果掉落的时间过长的话，我的记录的fall_time这一个变量会溢出，溢出是一个很重要的逻辑
```verilog
module top_module(
    input clk,
    input areset,    // Freshly brainwashed Lemmings walk left.
    input bump_left,
    input bump_right,
    input ground,
    input dig,
    output walk_left,
    output walk_right,
    output aaah,
    output digging ); 
    
    parameter LEFT=3'd0;
    parameter RIGHT=3'd1;
    parameter FALL_L=3'd2;
    parameter FALL_R=3'd3;
    parameter DIG_L=3'd4;
    parameter DIG_R=3'd5;
    parameter SPLAT=3'd6;
    
    reg [2:0]state,next_state;
    reg [4:0]fall_time;
    
    always @(*) begin
        if (!ground) begin
            if (state==LEFT || state==DIG_L) next_state=FALL_L;
            if (state==RIGHT || state==DIG_R) next_state=FALL_R;
        end
        else begin
            if ((state==FALL_L || state==FALL_R)&&(fall_time>5'd20)) next_state=SPLAT;
            if (dig) begin
                if (state==LEFT || state==FALL_L) next_state=DIG_L;
                if (state==RIGHT|| state==FALL_R) next_state=DIG_R;
            end
            else begin
                if (state==DIG_L || state==FALL_L) next_state=LEFT;
                if (state==DIG_R || state==FALL_R) next_state=RIGHT;
                if (state==LEFT && bump_left) next_state=RIGHT;
                if (state==RIGHT && bump_right) next_state=LEFT;
            end
        end
    end
    
    always @(posedge clk or posedge areset) begin
        if (areset) begin
            state<=LEFT;
            fall_time<=5'd0;
        end
        else begin
            state<=next_state;
            if (state==FALL_L || state==FALL_R) fall_time<=fall_time+5'd1;
            else fall_time<=5'd0;
        end
    end
    
    assign walk_left=(state==LEFT);
    assign walk_right=(state==RIGHT);
    assign aaah=(state==FALL_L || state==FALL_R);
    assign digging=(state==DIG_L || state==DIG_R);
                
    
endmodule


```



### Fsm ps2
![image.png](https://bucket-plusone.oss-cn-wuhan-lr.aliyuncs.com/img/20261101095249567.png)

这里也可以使用状态机的思路去解题，总体的思路还是一样的。也是声明当前的状态量和下一个时刻的状态量，然后去进行控制
```verilog
module top_module(
    input clk,
    input [7:0] in,
    input reset,    // Synchronous reset
    output done); //

    reg [1:0]state,next_state;
    // State transition logic (combinational)
    always @(*)begin
        case (state)
            2'b00:begin
                if (in[3]) next_state=2'b01;
                else next_state=2'b00;
            end
            2'b01:next_state=2'b10;
            2'b10:next_state=2'b11;
            2'b11:begin
                if (in[3]) next_state=2'b01;
                else next_state=2'b00;
            end
          
        endcase
    end
    // State flip-flops (sequential)
    always @(posedge clk)begin
        if (reset) begin
            state<=2'b00;
        end
        else begin
            state<=next_state;
        end
    end
    
    assign done=(state==2'b11);
            
    // Output logic

endmodule

```

### Fsm serial

![image.png](https://bucket-plusone.oss-cn-wuhan-lr.aliyuncs.com/img/20261101104235582.png)
这里面需要五个状态
idle：空闲，等待in变成0
data：计数器开始计数，结束时转向stop
stop状态，进行判断，看是不是合法的结束，合法的结束转向
done：接受成功，输出
error：接受失败，等待同步

```verilog
module top_module(
    input clk,
    input in,
    input reset,    // Synchronous reset
    output done
); 
    parameter IDLE  = 3'd0;
	parameter DATA  = 3'd1;
	parameter STOP  = 3'd2;
	parameter DONE  = 3'd3;
	parameter ERROR = 3'd4;

	reg [2:0] state, next_state;
	reg [2:0] cnt;
    always @(*)begin
        next_state=IDLE;
        case (state)
            IDLE:begin
                if (!in) next_state=DATA;
            end
            DATA:begin
                if (cnt<3'd7) next_state=DATA;
            	else next_state=STOP;
            end
            STOP:begin
                if (in) next_state=DONE;
                else next_state=ERROR;
            end
            DONE:begin
                if (!in) next_state=DATA;
                else begin
                	next_state=IDLE;
                end
            end
            ERROR:begin
                if (in) next_state=IDLE;
                else next_state=ERROR;
            end
        endcase
    end
  
   
	always @(posedge clk) begin
    	if (reset) begin
        	state <= IDLE;
        	cnt <= 3'd0;
    	end
    	else begin
        	state <= next_state;
 
	        if (state == DATA) begin
    	        if (cnt == 3'd7)
        	        cnt <= 3'd0;
            	else
                	cnt <= cnt + 3'd1;
        	end
        	else begin
            	cnt <= 3'd0;
        	end
    	end
	end
        
        
    assign done = (state == DONE);
                

endmodule

```

第二道题也差不多，就是加一个数据的储存，实际上是加上两句话
1.data[cnt]<=in;
2.assign out_byte=data;
还是蛮简单的，不用加入额外的逻辑什么的

### Fsm serialdp
![image.png](https://bucket-plusone.oss-cn-wuhan-lr.aliyuncs.com/img/20261101110121821.png)

```verilog

module top_module(
    input clk,
    input in,
    input reset,
    output [7:0] out_byte,
    output done
);

    // 定义五个状态
    parameter IDLE  = 3'd0;
    parameter DATA  = 3'd1;
    parameter STOP  = 3'd2;
    parameter DONE  = 3'd3;
    parameter ERROR = 3'd4;

    reg [2:0] state, next_state;
    reg [3:0] cnt;
    reg [7:0] data_reg;

    wire odd;
    wire parity_reset;

    // 奇校验模块
    assign parity_reset = reset || (state != DATA);

    parity p1 (
        .clk(clk),
        .reset(parity_reset),
        .in(in),
        .odd(odd)
    );

    // 1. 状态转移逻辑
    always @(*) begin
        next_state = state;

        case (state)
            IDLE: begin
                if (!in)
                    next_state = DATA;
                else
                    next_state = IDLE;
            end

            DATA: begin
                if (cnt == 4'd8)
                    next_state = STOP;
                else
                    next_state = DATA;
            end

            STOP: begin
                if (!in)
                    next_state = ERROR;
                else if (odd)
                    next_state = DONE;
                else
                    next_state = IDLE;
            end

            DONE: begin
                if (!in)
                    next_state = DATA;
                else
                    next_state = IDLE;
            end

            ERROR: begin
                if (in)
                    next_state = IDLE;
                else
                    next_state = ERROR;
            end

            default: next_state = IDLE;
        endcase
    end

    // 2. 状态寄存器、计数器、数据存储
    always @(posedge clk) begin
        if (reset) begin
            state <= IDLE;
            cnt <= 4'd0;
            data_reg <= 8'd0;
        end
        else begin
            state <= next_state;

            if (state == DATA) begin

                // 只保存前8个数据位
                if (cnt < 4'd8)
                    data_reg[cnt] <= in;

                // 接收9位：8数据位+1校验位
                if (cnt == 4'd8)
                    cnt <= 4'd0;
                else
                    cnt <= cnt + 4'd1;

            end
            else begin
                cnt <= 4'd0;
            end
        end
    end

    // 3. 输出逻辑
    assign done = (state == DONE);
    assign out_byte = data_reg;

endmodule
```
